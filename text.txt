        let transactions = [];

        // DOM elements
        const authContainer = document.getElementById('auth-container');
        const appContainer = document.getElementById('app-container');
        const authTabs = document.querySelectorAll('.auth-tab');
        const loginForm = document.getElementById('login-form');
        const signupForm = document.getElementById('signup-form');
        const loginBtn = document.getElementById('login-btn');
        const signupBtn = document.getElementById('signup-btn');
        const showSignupLink = document.getElementById('show-signup');
        const showLoginLink = document.getElementById('show-login');
        const logoutBtn = document.getElementById('logout-btn');
        const userAvatar = document.getElementById('user-avatar');
        const userName = document.getElementById('user-name');
        
        // Transaction elements
        const transactionForm = document.getElementById('transaction-form');
        const transactionList = document.getElementById('transaction-list');
        const totalIncomeEl = document.getElementById('total-income');
        const totalExpensesEl = document.getElementById('total-expenses');
        const balanceEl = document.getElementById('balance');
        
        // Graph elements
        const incomeExpenseChart = document.getElementById('income-expense-chart');
        const incomePercentEl = document.getElementById('income-percent');
        const incomeAmountEl = document.getElementById('income-amount');
        const expenseAmountEl = document.getElementById('expense-amount');
        
        const expenseCategoriesChart = document.getElementById('expense-categories-chart');
        const largestCategoryPercentEl = document.getElementById('largest-category-percent');
        const largestCategoryNameEl = document.getElementById('largest-category-name');
        const expenseCategoriesLegend = document.getElementById('expense-categories-legend');
        
        const categoryBarChart = document.getElementById('category-bar-chart');

        // Category colors for consistent visualization
        const categoryColors = {
            'food': '#FF6384',
            'transport': '#36A2EB',
            'entertainment': '#FFCE56',
            'bills': '#4BC0C0',
            'shopping': '#9966FF',
            'savings': '#2ecc71',
            'investment': '#e74c3c',
            'healthcare': '#FF9F40',
            'education': '#C9CBCF',
            'other': '#8c8c8c'
        };

        // Create particles for the amazing button
        function createButtonParticles() {
            const particlesContainer = document.getElementById('btn-particles');
            particlesContainer.innerHTML = '';
            
            for (let i = 0; i < 15; i++) {
                const particle = document.createElement('div');
                particle.classList.add('particle');
                
                // Random position
                const left = Math.random() * 100;
                const delay = Math.random() * 3;
                
                particle.style.left = `${left}%`;
                particle.style.animationDelay = `${delay}s`;
                
                particlesContainer.appendChild(particle);
            }
        }

        // Tab switching
        authTabs.forEach(tab => {
            tab.addEventListener('click', () => {
                const tabName = tab.getAttribute('data-tab');
                
                authTabs.forEach(t => t.classList.remove('active'));
                tab.classList.add('active');
                
                loginForm.classList.remove('active');
                signupForm.classList.remove('active');
                
                if (tabName === 'login') {
                    loginForm.classList.add('active');
                } else {
                    signupForm.classList.add('active');
                }
            });
        });

        // Show signup form
        showSignupLink.addEventListener('click', (e) => {
            e.preventDefault();
            authTabs[1].click();
        });

        // Show login form
        showLoginLink.addEventListener('click', (e) => {
            e.preventDefault();
            authTabs[0].click();
        });

        // Login function
        loginBtn.addEventListener('click', () => {
            const email = document.getElementById('login-email').value;
            const password = document.getElementById('login-password').value;
            
            if (!email || !password) {
                alert('Please fill in all fields');
                return;
            }
            
            // Simulate login - in a real app, this would be an API call
            const users = JSON.parse(localStorage.getItem('users')) || [];
            const user = users.find(u => u.email === email && u.password === password);
            
            if (user) {
                currentUser = user;
                localStorage.setItem('currentUser', JSON.stringify(currentUser));
                loadUserData();
                showApp();
            } else {
                alert('Invalid email or password');
            }
        });

        // Signup function
        signupBtn.addEventListener('click', () => {
            const firstName = document.getElementById('signup-firstname').value;
            const lastName = document.getElementById('signup-lastname').value;
            const email = document.getElementById('signup-email').value;
            const phone = document.getElementById('signup-phone').value;
            const password = document.getElementById('signup-password').value;
            
            if (!firstName || !lastName || !email || !phone || !password) {
                alert('Please fill in all fields');
                return;
            }
            
            // Check if user already exists
            const users = JSON.parse(localStorage.getItem('users')) || [];
            if (users.find(u => u.email === email)) {
                alert('User with this email already exists');
                return;
            }
            
            // Create new user
            const newUser = {
                id: generateID(),
                firstName,
                lastName,
                email,
                phone,
                password,
                loginMethod: 'email'
            };
            
            users.push(newUser);
            localStorage.setItem('users', JSON.stringify(users));
            
            currentUser = newUser;
            localStorage.setItem('currentUser', JSON.stringify(currentUser));
            
            loadUserData();
            showApp();
            
            // Switch to login tab for next time
            authTabs[0].click();
        });

        // Logout function
        logoutBtn.addEventListener('click', () => {
            currentUser = null;
            localStorage.removeItem('currentUser');
            authContainer.style.display = 'block';
            appContainer.style.display = 'none';
            
            // Clear forms
            document.getElementById('login-email').value = '';
            document.getElementById('login-password').value = '';
            document.getElementById('signup-firstname').value = '';
            document.getElementById('signup-lastname').value = '';
            document.getElementById('signup-email').value = '';
            document.getElementById('signup-phone').value = '';
            document.getElementById('signup-password').value = '';
        });

        // Show main app
        function showApp() {
            authContainer.style.display = 'none';
            appContainer.style.display = 'block';
            createButtonParticles();
        }

        // Load user data
        function loadUserData() {
            // Update user info in header
            userAvatar.textContent = currentUser.firstName.charAt(0).toUpperCase();
            userName.textContent = `${currentUser.firstName} ${currentUser.lastName}`;
            
            // Load user's transactions
            const userTransactions = JSON.parse(localStorage.getItem(`transactions_${currentUser.id}`)) || [];
            transactions = userTransactions;
            
            updateUI();
        }

        // Generate unique ID
        function generateID() {
            return Math.floor(Math.random() * 1000000);
        }

        // Add transaction
        transactionForm.addEventListener('submit', function(e) {
            e.preventDefault();
            
            const description = document.getElementById('description').value;
            const amount = parseFloat(document.getElementById('amount').value);
            const type = document.getElementById('type').value;
            const category = document.getElementById('category').value;
            
            if (!description || !amount || !type || !category) {
                alert('Please fill in all fields');
                return;
            }
            
            const transaction = {
                id: generateID(),
                description,
                amount,
                type,
                category,
                date: new Date().toLocaleDateString()
            };
            
            transactions.push(transaction);
            saveUserData();
            updateUI();
            transactionForm.reset();
            
            // Button animation feedback
            const submitBtn = document.querySelector('.submit-btn');
            submitBtn.style.background = 'linear-gradient(135deg, #2ecc71, #27ae60)';
            setTimeout(() => {
                submitBtn.style.background = 'linear-gradient(135deg, #6a11cb 0%, #2575fc 100%)';
            }, 500);
        });

        // Remove transaction
        function removeTransaction(id) {
            transactions = transactions.filter(transaction => transaction.id !== id);
            saveUserData();
            updateUI();
        }

        // Save user data to localStorage
        function saveUserData() {
            localStorage.setItem(`transactions_${currentUser.id}`, JSON.stringify(transactions));
        }

        // Update UI
        function updateUI() {
            // Clear transaction list
            transactionList.innerHTML = '';
            
            if (transactions.length === 0) {
                transactionList.innerHTML = '<div class="transaction-item" style="text-align: center; color: #777; padding: 20px;">No transactions yet. Add your first transaction above.</div>';
            } else {
                // Add transactions to list
                transactions.forEach(transaction => {
                    const transactionEl = document.createElement('div');
                    transactionEl.classList.add('transaction-item');
                    transactionEl.classList.add(transaction.type === 'income' ? 'income-transaction' : 'expense-transaction');
                    
                    transactionEl.innerHTML = `
                        <div class="transaction-details">
                            <span class="transaction-name">${transaction.description}</span>
                            <span class="transaction-category">${transaction.category} • ${transaction.date}</span>
                        </div>
                        <div style="display: flex; align-items: center; gap: 10px;">
                            <span class="transaction-amount">${transaction.type === 'income' ? '+' : '-'}₹${transaction.amount.toFixed(2)}</span>
                            <button class="delete-btn" onclick="removeTransaction(${transaction.id})">Delete</button>
                        </div>
                    `;
                    
                    transactionList.appendChild(transactionEl);
                });
            }
            
            // Update summary
            updateSummary();
            
            // Update graphs
            updateIncomeExpenseChart();
            updateExpenseCategoriesChart();
            updateCategoryBarChart();
        }

        // Update summary
        function updateSummary() {
            const incomes = transactions
                .filter(transaction => transaction.type === 'income')
                .reduce((total, transaction) => total + transaction.amount, 0);
            
            const expenses = transactions
                .filter(transaction => transaction.type === 'expense')
                .reduce((total, transaction) => total + transaction.amount, 0);
            
            const balance = incomes - expenses;
            
            totalIncomeEl.textContent = `₹${incomes.toFixed(2)}`;
            totalExpensesEl.textContent = `₹${expenses.toFixed(2)}`;
            balanceEl.textContent = `₹${balance.toFixed(2)}`;
        }

        // Update Income vs Expense Pie Chart
        function updateIncomeExpenseChart() {
            const incomes = transactions
                .filter(transaction => transaction.type === 'income')
                .reduce((total, transaction) => total + transaction.amount, 0);
            
            const expenses = transactions
                .filter(transaction => transaction.type === 'expense')
                .reduce((total, transaction) => total + transaction.amount, 0);
            
            const total = incomes + expenses;
            
            // Calculate percentages
            let incomePercent = 0;
            let expensePercent = 0;
            
            if (total > 0) {
                incomePercent = (incomes / total) * 100;
                expensePercent = (expenses / total) * 100;
            }
            
            // Update pie chart
            incomeExpenseChart.style.background = `conic-gradient(
                #2ecc71 0% ${incomePercent}%, 
                #e74c3c ${incomePercent}% 100%
            )`;
            
            // Update display values
            incomePercentEl.textContent = `${incomePercent.toFixed(1)}%`;
            incomeAmountEl.textContent = `₹${incomes.toFixed(2)}`;
            expenseAmountEl.textContent = `₹${expenses.toFixed(2)}`;
        }

        // Update Expense Categories Pie Chart
        function updateExpenseCategoriesChart() {
            // Get expenses by category
            const expensesByCategory = {};
            transactions
                .filter(transaction => transaction.type === 'expense')
                .forEach(transaction => {
                    if (!expensesByCategory[transaction.category]) {
                        expensesByCategory[transaction.category] = 0;
                    }
                    expensesByCategory[transaction.category] += transaction.amount;
                });
            
            // Calculate total expenses
            const totalExpenses = Object.values(expensesByCategory).reduce((sum, amount) => sum + amount, 0);
            
            // Clear legend
            expenseCategoriesLegend.innerHTML = '';
            
            if (totalExpenses === 0) {
                // No expenses yet
                expenseCategoriesChart.style.background = '#f0f0f0';
                largestCategoryPercentEl.textContent = '0%';
                largestCategoryNameEl.textContent = 'No Data';
                return;
            }
            
            // Sort categories by amount (descending)
            const sortedCategories = Object.entries(expensesByCategory)
                .sort((a, b) => b[1] - a[1]);
            
            // Generate conic gradient for pie chart
            let gradientString = '';
            let accumulatedPercent = 0;
            
            sortedCategories.forEach(([category, amount], index) => {
                const percent = (amount / totalExpenses) * 100;
                const color = categoryColors[category] || '#8c8c8c';
                
                gradientString += `${color} ${accumulatedPercent}% ${accumulatedPercent + percent}%`;
                if (index < sortedCategories.length - 1) {
                    gradientString += ', ';
                }
                
                accumulatedPercent += percent;
            });
            
            expenseCategoriesChart.style.background = `conic-gradient(${gradientString})`;
            
            // Update center display with largest category
            const largestCategory = sortedCategories[0];
            if (largestCategory) {
                const largestPercent = (largestCategory[1] / totalExpenses) * 100;
                largestCategoryPercentEl.textContent = `${largestPercent.toFixed(1)}%`;
                largestCategoryNameEl.textContent = largestCategory[0].charAt(0).toUpperCase() + largestCategory[0].slice(1);
            }
            
            // Update legend
            sortedCategories.forEach(([category, amount]) => {
                const percent = (amount / totalExpenses) * 100;
                const color = categoryColors[category] || '#8c8c8c';
                
                const legendItem = document.createElement('div');
                legendItem.classList.add('legend-item');
                
                legendItem.innerHTML = `
                    <div class="legend-info">
                        <div class="legend-color" style="background-color: ${color};"></div>
                        <span class="legend-label">${category.charAt(0).toUpperCase() + category.slice(1)}</span>
                    </div>
                    <span class="legend-value">${percent.toFixed(1)}% (₹${amount.toFixed(2)})</span>
                `;
                
                expenseCategoriesLegend.appendChild(legendItem);
            });
        }

        // Update Category Bar Chart
        function updateCategoryBarChart() {
            // Get expenses by category
            const expensesByCategory = {};
            transactions
                .filter(transaction => transaction.type === 'expense')
                .forEach(transaction => {
                    if (!expensesByCategory[transaction.category]) {
                        expensesByCategory[transaction.category] = 0;
                    }
                    expensesByCategory[transaction.category] += transaction.amount;
                });
            
            // Calculate total expenses
            const totalExpenses = Object.values(expensesByCategory).reduce((sum, amount) => sum + amount, 0);
            
            // Clear bar chart
            categoryBarChart.innerHTML = '';
            
            if (totalExpenses === 0) {
                // No expenses yet
                const noDataMsg = document.createElement('div');
                noDataMsg.textContent = 'No expense data available';
                noDataMsg.style.textAlign = 'center';
                noDataMsg.style.color = '#777';
                noDataMsg.style.marginTop = '80px';
                categoryBarChart.appendChild(noDataMsg);
                return;
            }
            
            // Find maximum expense for scaling
            const maxExpense = Math.max(...Object.values(expensesByCategory));
            
            // Create bars for each category
            Object.entries(expensesByCategory).forEach(([category, amount]) => {
                const percent = (amount / totalExpenses) * 100;
                const barHeight = (amount / maxExpense) * 100;
                const color = categoryColors[category] || '#8c8c8c';
                
                const barContainer = document.createElement('div');
                barContainer.classList.add('bar-container');
                
                barContainer.innerHTML = `
                    <div class="bar" style="height: ${barHeight}%; background-color: ${color};">
                        <div class="bar-value">${percent.toFixed(1)}%</div>
                    </div>
                    <div class="bar-label">${category.charAt(0).toUpperCase() + category.slice(1)}</div>
                `;
                
                categoryBarChart.appendChild(barContainer);
            });
        }

        // Check if user is already logged in
        function checkLoginStatus() {
            const loggedInUser = localStorage.getItem('currentUser');
            if (loggedInUser) {
                currentUser = JSON.parse(loggedInUser);
                loadUserData();
                showApp();
            }
        }

        // Create dynamic background
        const background = document.getElementById('dynamicBackground');
        const colors = [
            'linear-gradient(45deg, #0c0c0c, #142850)',
            'linear-gradient(45deg, #0c0c0c, #27496d)',
            'linear-gradient(45deg, #0c0c0c, #0c7b93)',
            'linear-gradient(45deg, #0c0c0c, #00a8cc)'
        ];
        
        let currentIndex = 0;
        
        function changeBackground() {
            background.style.background = colors[currentIndex];
            background.style.transition = 'background 2s ease';
            
            currentIndex = (currentIndex + 1) % colors.length;
        }
        
        // Initial call
        changeBackground();
        
        // Change background every 5 seconds
        setInterval(changeBackground, 5000);
        
        // Add floating particles to background
        function createParticles() {
            for (let i = 0; i < 50; i++) {
                const particle = document.createElement('div');
                particle.style.position = 'absolute';
                particle.style.width = Math.random() * 5 + 2 + 'px';
                particle.style.height = particle.style.width;
                particle.style.background = 'rgba(255, 138, 0, 0.3)';
                particle.style.borderRadius = '50%';
                particle.style.top = Math.random() * 100 + 'vh';
                particle.style.left = Math.random() * 100 + 'vw';
                particle.style.boxShadow = '0 0 10px rgba(255, 138, 0, 0.5)';
                background.appendChild(particle);
                
                // Animate particles
                animateParticle(particle);
            }
        }
        
        function animateParticle(particle) {
            let x = Math.random() * 100;
            let y = Math.random() * 100;
            let duration = Math.random() * 20 + 10;
            
            particle.animate([
                { transform: `translate(${x}vw, ${y}vh)` }
            ], {
                duration: duration * 1000,
                iterations: Infinity,
                direction: 'alternate',
                easing: 'ease-in-out'
            });
        }
        
        createParticles();

        // Initialize the app
        checkLoginStatus();
    </script>
